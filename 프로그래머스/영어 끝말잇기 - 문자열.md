- 코드 작성 ( 1차 아무 도움 없이 작성 ) ( 시간 : 40m )
  ```cpp
  #include <string>
  #include <vector>
  #include <iostream>

  using namespace std;

  struct person {
      int idx;
      vector<string> wd;
  };

  vector<int> solution(int n, vector<string> words) {
      vector<int> answer;
      // 사람 수가 있음. ( % 이걸로 뭔가 해도 될 듯 )
      // 앞 뒤를 비교해야 함
      // 각 사람마다 단어를 넣어서 기록해야 함.
      // 사람마다도 번호가 있고, 1부터 n까지 임.
      // 구조체에 벡터, 번호 넣기
      // 입력
      // 뒷 끝과 앞 끝이 같은 단어의 연속 벡터
      // 출력
      // [가장 먼저 탈락하는 사람 번호, 탈락자가 탈락하는 차례]
      // 탈락자 안 나오면 [0, 0]
      // 탈락 기준 : 중복 단어 뱉음 or 끝말잇기 실패
      vector<person> people;
      int count = 0;

      for (int i = 1; i <= n; i++) {
          person p;
          p.idx = i;
          people.push_back(p);
      }

      string first_w = words[0];

      for (int i = 1; i < words.size(); i++) {
          // words.size는 벡터 인덱스에 써서
          // 전체 갯수에서 -1 이여야 함.
          if (i == 1) {
              count++;
              if (first_w.back() == words[i].front()) {
                  people[0].wd.push_back(first_w);
                  people[1].wd.push_back(words[1]);
                  first_w = words[1];
              } else {
                  answer.push_back(2); // 사람 번호
                  answer.push_back(1); // 그 인간의 단어 갯수
              }
          } else {
              count++;
              if (first_w.back() == words[i].front()) {
                  people[i - 1].wd.push_back(first_w);
                  people[i].wd.push_back(words[i]);
                  first_w = words[i];
              } else {
                  int num;
                  if (count % n == 0) {
                      num = n;
                  } else {
                      num = count % n;
                  }
                  answer.push_back(num);
                  answer.push_back(people[num].wd.size());
              }
          }
      }
      return answer;
  }
  ```
- 답안 추출 ( 더 효율적 방식 → 흡수 ( top-down ) )
  ```cpp
  #include <string>
  #include <vector>
  #include <iostream>
  #include <set>

  using namespace std;

  vector<int> solution(int n, vector<string> words) {
      // 이미 등장한 단어를 저장할 집합
      set<string> used_words;

      // 첫 번째 단어는 미리 넣어둡니다.
      used_words.insert(words[0]);

      for (int i = 1; i < words.size(); i++) {
          string prev = words[i - 1];
          string curr = words[i];

          // 1. 끝말잇기 규칙 확인 (이전 단어 끝문자 != 현재 단어 첫문자)
          // 2. 중복 단어 확인 (이미 set에 존재하는 경우)
          if (prev.back() != curr.front() || used_words.find(curr) != used_words.end()) {
              int person_num = (i % n) + 1;
              int turn = (i / n) + 1;
              // 모듈러 쓰는 공식처럼 외워놔라. ( 사람 번호는 '모듈러 + 1' )
              // 차례는 '현재 총 진행 횟수 / 사람 수 + 1'
              return {person_num, turn};
          }

          // 통과했다면 사용한 단어 목록에 추가
          used_words.insert(curr);
      }

      // 탈락자가 없으면 [0, 0] 반환
      return {0, 0};
  }
  ```
