# 중요한 단어를 스포 방지

- 문제 파악
  - 입력
    - 그냥 문장 ( 띄어쓰기 포함 )
    - 소문자, 숫자로만 구성
    - 범위 (두 단어에 걸치고 있을 수도 있음)
    - 오름 차순으로 주어짐
    - 공백은 인덱스에 안 넣는듯 (공백을 뺀 범위를 만들어야 함.)
  - 출력
    - 중요한 단어 갯수 출력
  - 방법
    - 단어를 나누는 기준을 띄어쓰기로 하자. 한 문장으로 쭉 주니까.
- 코드
  - 1차 정답 코드 ( 찐 정답은 아닌데 방향성이 이 쪽 ) → 0점
    ```cpp
    #include <string>
    #include <vector>

    using namespace std;

    int solution(string message, vector<vector<int>> spoiler_ranges) {
        int answer = 0;
        vector<int> char_to_word_index;
        int current_word_idx = 0;
        bool in_word = false;

        for (char c : message) {
            if (c == ' ') {
                // 공백을 만나면 단어가 끝났음을 표시
                if (in_word) {
                    current_word_idx++;
                    in_word = false;
                }
            } else {
                // 공백이 아닌 문자를 만나면 현재 단어 인덱스를 기록
                in_word = true;
                char_to_word_index.push_back(current_word_idx);
            }
        }

        int total_words = current_word_idx;
        if (in_word) total_words++;

        vector<bool> is_spoiled(total_words, false);

        for (const auto& range : spoiler_ranges) {
            int start = range[0];
            int end = range[1];
            // 주어진 범위 내의 인덱스를 순회하며 해당 단어를 스포일러 표시
            for (int i = start; i < end; ++i) {
                if (i >= 0 && i < char_to_word_index.size()) {
                    int target_word = char_to_word_index[i];
                    is_spoiled[target_word] = true;
                }
            }
        }

        for (bool spoiled : is_spoiled) {
            if (spoiled) {
                answer++;
            }
        }

        return answer;
    }
    ```

            }

        vector<bool> is_spoiled(total_words, false);

            int start = range[0];
            int end = range[1];

            for (int i = start; i <= end; ++i) {
                if (i >= 0 && i < char_to_word_index.size()) {
                    int target_word = char_to_word_index[i];
                    is_spoiled[target_word] = true;
                }
            }
        }

        // 3. 스포일러 처리된 단어의 개수를 합산
        int answer = 0;
        for (bool spoiled : is_spoiled) {
            if (spoiled) {
                answer++;
            }
        }

        return answer;
    }

  - 2차 코드 ( 두 단어에 걸치는 경우 어떻게 해야하나 ) → 58점
    - 0점이었던 것은 오류가 아니라 걍 최소 테스트케이스도 통과 못했다는 의미이다.
    ```cpp
    #include <string>
    #include <vector>
    #include <set>
    #include <map>

    using namespace std;

    struct WordInfo {
        string text;
        int start;
        int end;
        bool touched_spoiler = false;
        bool touched_safe_zone = false;
    };

    int solution(string message, vector<vector<int>> spoiler_ranges) {
        int n = message.size();
        vector<bool> is_spoiled_idx(n, false);

        // 1. 스포일러가 적용된 모든 인덱스를 마킹
        for (const auto& range : spoiler_ranges) {
            for (int i = range[0]; i <= range[1]; ++i) {
                if (i >= 0 && i < n) {
                    is_spoiled_idx[i] = true;
                }
                // a---- (스포 : (1, 4))
                // abdad asd ffdg : message 길이 대충 17
                // 아, 메시지 길이 안이면 그 인덱스에 넣으라는 뜻이구만.
            }
        }

        // 2. 단어 단위로 분리하며 각 단어의 상태 확인
        vector<WordInfo> all_words;
        string temp = "";
        int start_pos = -1;

        for (int i = 0; i <= n; ++i) {
            if (i < n && message[i] != ' ') {
                if (start_pos == -1) start_pos = i;
                temp += message[i];
            } else {
                if (!temp.empty()) {
                    // 공백인 경우까지 앞의 if로 처리하고, string temp에 하나씩 더해서 단어를 생성함.
                    WordInfo word;
                    word.text = temp;
                    word.start = start_pos;
                    word.end = i - 1;

                    // 단어의 각 문자를 돌며 스포일러 구역과 안전 구역 접촉 여부 확인
                    for (int j = word.start; j <= word.end; ++j) {
                        if (is_spoiled_idx[j]) word.touched_spoiler = true;
                        else word.touched_safe_zone = true;
                    }
                    all_words.push_back(word);
                    temp = "";
                    start_pos = -1; // 초기화하고 다음 단어 탐색
                }
            }
        }

        // 3. 특정 단어 텍스트가 "한 번이라도" 안전 구역에 등장했는지 기록
        // 규칙: 스포 방지 구간이 아닌 구간에 등장한 적이 없어야 함
        map<string, bool> is_forbidden;
        for (const auto& w : all_words) {
            if (w.touched_safe_zone) {
                is_forbidden[w.text] = true;
            }
        }

        // 4. 왼쪽 단어부터 순서대로 중요한 단어인지 판단
        set<string> important_words;
        int count = 0;

        for (const auto& w : all_words) {
            // 조건 1: 스포일러 구간에 걸쳐 있음
            // 조건 2: 메시지 내의 어떤 안전 구간(비 스포일러 구간)에도 등장하지 않음
            // 조건 3: 이전에 이미 중요한 단어로 등록되지 않음
            if (w.touched_spoiler && !is_forbidden[w.text]) {
                if (important_words.find(w.text) == important_words.end()) {
                    important_words.insert(w.text);
                    count++;
                }
            }
        }

        return count;
    }
    ```
