  ```cpp
                      for (int j = 1; j <= y; j++) {
                          signals_time[k].push_back(signals_time[k][i - 1] + (g + r) + j);
                      }
                  }
              }
          }

          bool pass = true;
          int box;
          for (int i = 0; i < 5; i++) { // 5번 반복
  ```

- 2차 수정 ( % 이용 코드 작성 )
  ```cpp
  #include <string>
  #include <vector>

  using namespace std;

  int solution(vector<vector<int>> signals) {
      int answer = 0;
      int n = int(signals.size());
      vector<int> period;
      int g, y, r;

      for (int i = 0; i < n; i++) {
          g = signals[i][0];
          y = signals[i][1];
          r = signals[i][2];
          period.push_back(g + y + r);
      }

      int start = 0;
      int count = 0;
      bool pass = true;

      for (int t = 1; t < 1000; t++) {
          count++;
          // 나머지가 g 이상, r 이하이면 그게 y에 해당하는 범위라는 뜻임. t가 1인 점에 대해서. 모두 그 조건으로 조사.
          for (int i = 0; i < n; i++) {
              start = t % period[i];
              if (signals[i][0] < start <= signals[i][0] + signals[i][1]) {
                  continue;
              } else {
                  pass = false;
                  break;
              }
              if (pass) {
                  answer = start;
                  break;
              }
          }
          if (answer != 0) {
              break;
          }
      }

      if (count == 1000) {
          return -1;
      }
      return answer;
  }
  ```

- 교훈
  - 한 문제를 여러번 풀면 그 문제 유형 혹은 문제 풀이 자체에 대한 어떤 패턴 블록을 흡수할 수 있다. 그래서 푸는 시간이 비약적으로 줄어든다.
  - 유형이 뭔지 알면, 접근이 매우 빨라진다. 정말 필요한 것만 입력할 수 있다. 마치 레고 조립 처럼. 그리고 그것을 수집하려면 올바른 답지와 풀이를 갖고 있어야 한다. 스스로 고민하는 것은 그런 블록이 있을 때의 일이다. 블록이 없는 경우는 한 10분 고민하다 블록의 설명서를 보고 그것을 흡수하는 것이 우선이다.
  - 필요한 부분에만 헬스 PT받은 것처럼 힘을 줄 수 있다.
  - Bool을 쓰는 것이 count를 갖고 하는 것보다 훨씬 낫다는 것.
  - 2중 for문은 지양하는게 낫고, for을 각각 분리시킬 수 있으면 분리시키는게 훨씬 디버깅이 편하다는 것
